function hashStr(str) {
    const coreStr = String(str || "").replace(/\s+/g, " ").trim().slice(0, 300);
    let hash = 0;
    for (let i = 0; i < coreStr.length; i++) {
        hash = Math.imul(31, hash) + coreStr.charCodeAt(i) | 0;
    }
    return Math.abs(hash).toString(16);
}

export function verifyTextWithCode(text, code) {
    const codeAt = text.indexOf(code);
    if (codeAt < 0) {
        throw new Error(`[CORE] 验证码实体定位失败: ${String(code || "").slice(0, 32)}`);
    }
    return {
        text,
        entities: [{ type: "code", offset: codeAt, length: code.length }]
    };
}

}

function isCommand(cmd, aliases) {
    return aliases.includes(String(cmd || "").toLowerCase());
}

function pickRandomChar(chars) {
    const bytes = new Uint32Array(1);
    crypto.getRandomValues(bytes);
    return chars[bytes[0] % chars.length];
}

function shuffleChars(chars) {
    const arr = [...chars];
    const bytes = new Uint32Array(arr.length);
    crypto.getRandomValues(bytes);

    for (let i = arr.length - 1; i > 0; i--) {
        const j = bytes[i] % (i + 1);
        [arr[i], arr[j]] = [arr[j], arr[i]];
    }

    return arr.join("");
}

function makeVerifyCode() {
    const upperChars = "ABCDEFGHJKLMNPQRSTUVWXYZ";
    const lowerChars = "abcdefghijkmnpqrstuvwxyz";
    const numberChars = "23456789";
    const symbolChars = "!$%&_-";
    const allChars = upperChars + lowerChars + numberChars + symbolChars;

    const chars = [
        pickRandomChar(upperChars),
        pickRandomChar(lowerChars),
        pickRandomChar(numberChars),
        pickRandomChar(symbolChars),
        pickRandomChar(allChars),
        pickRandomChar(allChars)
    ];

    return shuffleChars(chars);
}


function parseVerifyState(raw) {
    if (!raw) {
        return { ok: false, donePrompted: false };
    }

    try {
        const parsed = JSON.parse(raw);
        return {
            ok: !!parsed?.ok,
            donePrompted: !!parsed?.donePrompted
        };
    } catch (e) {
        return { ok: false, donePrompted: false };
    }
}

function isVerifiedState(state) {
    return !!state?.ok;
}

function serializeVerifyState(state) {
    return JSON.stringify({
        ok: !!state?.ok,
        donePrompted: !!state?.donePrompted
    });
}

function parseVerifyWaitState(raw) {
    if (!raw) return null;

    try {
        const parsed = JSON.parse(raw);
        return {
            startWarned: !!parsed?.startWarned,
            wrongWarnCount: Math.max(0, Number(parsed?.wrongWarnCount || 0) || 0),
            expiredWarnCount: Math.max(0, Number(parsed?.expiredWarnCount || 0) || 0),
            code: normalizeVerifyCode(parsed?.code || ""),
            expiresAt: Number(parsed?.expiresAt || 0)
        };
    } catch (e) {
        return {
            startWarned: false,
            wrongWarnCount: 0,
            expiredWarnCount: 0,
            code: "",
            expiresAt: 0
        };
    }
}

function isVerifyWaitActive(state) {
    if (!state) return false;
    if (!state.expiresAt) return true;
    return Date.now() <= state.expiresAt;
}

function buildVerifyWaitState(
    code = "",
    expiresAt = Date.now() + TTL.VERIFY_WINDOW * 1000,
    startWarned = false,
    wrongWarnCount = 0,
    expiredWarnCount = 0
) {
    return {
        startWarned: !!startWarned,
        wrongWarnCount: Math.max(0, Number(wrongWarnCount || 0) || 0),
        expiredWarnCount: Math.max(0, Number(expiredWarnCount || 0) || 0),
        code: normalizeVerifyCode(code),
        expiresAt: Number(expiresAt || 0)
    };
}

function serializeVerifyWaitState(state) {
    return JSON.stringify(buildVerifyWaitState(
        state?.code,
        state?.expiresAt,
        state?.startWarned,
        state?.wrongWarnCount,
        state?.expiredWarnCount
    ));
}

const HOT_UPDATE_CACHE = new Map();
const RATE_STATE_CACHE = new Map();
const VERIFY_NEED_START_CACHE = new Map();

function cleanupVerifyNeedStartCache(now = Date.now()) {
    for (const [key, value] of VERIFY_NEED_START_CACHE) {
        if (!value || (now - Number(value.ts || 0)) > TTL.VERIFY_WINDOW * 1000) {
            VERIFY_NEED_START_CACHE.delete(key);
        }
    }
}

function shouldSendNeedStart(uid) {
    const key = String(uid || "").trim();
    if (!key) return false;

    const now = Date.now();
    cleanupVerifyNeedStartCache(now);

    const old = VERIFY_NEED_START_CACHE.get(key) || { count: 0, ts: now };
    const count = Number(old.count || 0);
    VERIFY_NEED_START_CACHE.set(key, { count: count + 1, ts: now });

    return count < 1;
}

function clearNeedStartWarn(uid) {
    const key = String(uid || "").trim();
    if (key) VERIFY_NEED_START_CACHE.delete(key);
}

function cleanupHotUpdateCache(now = Date.now()) {
    for (const [key, value] of HOT_UPDATE_CACHE) {
        if (!value || (now - Number(value.ts || 0)) > HOT_UPDATE_TTL_MS) {
            HOT_UPDATE_CACHE.delete(key);
        }
    }

    if (HOT_UPDATE_CACHE.size <= HOT_UPDATE_CACHE_LIMIT) return;

    const entries = Array.from(HOT_UPDATE_CACHE.entries()).sort(
        (a, b) => Number(a[1]?.ts || 0) - Number(b[1]?.ts || 0)
    );
    const overflow = HOT_UPDATE_CACHE.size - HOT_UPDATE_CACHE_LIMIT;

    for (let i = 0; i < overflow; i++) {
        HOT_UPDATE_CACHE.delete(entries[i][0]);
    }
}

function getHotUpdateState(updateId) {
    const key = String(updateId ?? "").trim();
    if (!key) return "";

    const now = Date.now();
    cleanupHotUpdateCache(now);

    const entry = HOT_UPDATE_CACHE.get(key);
    if (!entry) return "";

    if ((now - Number(entry.ts || 0)) > HOT_UPDATE_TTL_MS) {
        HOT_UPDATE_CACHE.delete(key);
        return "";
    }

    return String(entry.state || "");
}

function setHotUpdateState(updateId, state) {
    const key = String(updateId ?? "").trim();
    if (!key) return;

    const now = Date.now();
    HOT_UPDATE_CACHE.set(key, { state: String(state || ""), ts: now });
    cleanupHotUpdateCache(now);
}

function clearHotUpdateState(updateId) {
    const key = String(updateId ?? "").trim();
    if (!key) return;
    HOT_UPDATE_CACHE.delete(key);
}

function cleanupRateStateCache(now = Date.now()) {
    for (const [key, value] of RATE_STATE_CACHE) {
        if (!value || (now - Number(value.touched || value.start || 0)) > RATE_STATE_CACHE_TTL_MS) {
            RATE_STATE_CACHE.delete(key);
        }
    }

    if (RATE_STATE_CACHE.size <= RATE_STATE_CACHE_LIMIT) return;

    const entries = Array.from(RATE_STATE_CACHE.entries()).sort(
        (a, b) => Number(a[1]?.touched || a[1]?.start || 0) - Number(b[1]?.touched || b[1]?.start || 0)
    );
    const overflow = RATE_STATE_CACHE.size - RATE_STATE_CACHE_LIMIT;

    for (let i = 0; i < overflow; i++) {
        RATE_STATE_CACHE.delete(entries[i][0]);
    }
}

function errorLogPrefix() {
    return KV_PREFIX.SYS + "errlog_";
}

function errorLogEntryKey(level, text) {
    return errorLogPrefix() + Date.now() + "_" + hashStr(`${level}:${text}`) + "_" + Math.random().toString(36).slice(2, 6);
}

export default {
    async fetch(request, env, ctx) {
        const { BOT_TOKEN, MY_ID, ADMIN_UID, SECRET_TOKEN, KV } = env;

        if (!KV) {
            return new Response("KV binding missing", { status: 500 });
        }

        if (request.method !== "POST") {
            return new Response("RelayBot Standby");
        }

        function sleep(ms) {
            return new Promise((resolve) => setTimeout(resolve, ms));
        }

        async function tg(method, params, retries = TG_MAX_RETRIES) {
            const controller = new AbortController();
            const timer = setTimeout(() => controller.abort(), TG_TIMEOUT_MS);

            try {
                const res = await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/${method}`, {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify(params),
                    signal: controller.signal
                });

                const raw = await res.text();
                let json;
                try {
                    json = JSON.parse(raw);
                } catch (e) {
                    return { ok: false, description: `Telegram 非 JSON 响应 (HTTP ${res.status})` };
                }

                if (!json.ok && Number(json.error_code || 0) === 429 && retries > 0) {
                    const waitMs = Math.max(1, Number(json.parameters?.retry_after || 5)) * 1000;
                    await sleep(waitMs);
                    return tg(method, params, retries - 1);
                }

                return json;
            } catch (e) {
                if (e?.name === "AbortError") {
                    return { ok: false, description: `Fetch超时(${TG_TIMEOUT_MS}ms)` };
                }
                return { ok: false, description: "Fetch网络断开或超时" };
            } finally {
                clearTimeout(timer);
            }
        }

        function normalizeThreadId(threadId) {
            const raw = String(threadId ?? "").trim();
            if (!/^\d+$/.test(raw)) return "";
            const n = Number(raw);
            if (!Number.isSafeInteger(n) || n <= 0) return "";
            return String(n);
        }

        function isValidThreadId(threadId) {
            return !!normalizeThreadId(threadId);
        }

        function withRequiredThread(base, threadId, context = "Topic") {
            const tid = normalizeThreadId(threadId);
            if (!tid) {
                throw new Error(`[CORE] ${context} 缺少有效 Topic ID，已阻止消息落入总控制室`);
            }
            return { ...base, message_thread_id: Number(tid) };
        }

        async function assertForwardLandedInThread(res, expectedThreadId, uid, phase) {
            if (!res?.ok) return;
            const expected = normalizeThreadId(expectedThreadId);
            const actual = normalizeThreadId(res?.result?.message_thread_id);

            if (!expected || actual === expected) return;

            const wrongMsgId = res?.result?.message_id;
            if (wrongMsgId) {
                ctx.waitUntil(deleteMessageNow(MY_ID, wrongMsgId));
            }

            throw new Error(`[CORE] ${phase}(UID:${uid})疑似落入总控制室，系统已阻止继续处理`);
        }

        function parseErr(res) {
            const desc = String(res?.description || "未知错误");
            for (const rule of ERR_ROUTER) {
                if (desc.includes(rule.match)) return { action: rule.action, desc };
            }
            const code = Number(res?.error_code || 0);
            return { action: (code === 400 || code === 403) ? "TERMINAL" : "CORE", desc };
        }

        async function deleteMessageNow(chatId, messageId) {
            if (!chatId || !messageId) return;
            try {
                await tg("deleteMessage", {
                    chat_id: chatId,
                    message_id: messageId
                });
            } catch (e) {
                // best-effort
            }
        }

        async function deleteTopicNow(threadId) {
            if (!threadId) return;
            try {
                await tg("deleteForumTopic", {
                    chat_id: MY_ID,
                    message_thread_id: Number(threadId)
                });
            } catch (e) {
                // best-effort
            }
        }

        function deleteLater(chatId, messageId, delaySeconds) {
            if (!chatId || !messageId || !delaySeconds) return;
            ctx.waitUntil((async () => {
                try {
                    await sleep(delaySeconds * 1000);
                    await tg("deleteMessage", {
                        chat_id: chatId,
                        message_id: messageId
                    });
                } catch (e) {
                    // best-effort
                }
            })());
        }

        async function safeEditMessageText(chatId, messageId, text, replyMarkup) {
            const params = {
                chat_id: chatId,
                message_id: messageId,
                text
            };
            if (replyMarkup !== undefined) {
                params.reply_markup = replyMarkup;
            }
            return tg("editMessageText", params);
        }

        async function sendControlMessage(params, deleteAfterSeconds = 0) {
            const res = await tg("sendMessage", params);
            if (res.ok && res.result?.message_id && deleteAfterSeconds > 0) {
                deleteLater(params.chat_id, res.result.message_id, deleteAfterSeconds);
            }
            return res;
        }

        async function isGuestRateLimited(uid) {
            const key = String(uid || "").trim();
            if (!key) return false;

            const now = Date.now();
            cleanupRateStateCache(now);

            let nextState = { start: now, count: 1, warned: false, touched: now };
            const oldState = RATE_STATE_CACHE.get(key);

            if (oldState) {
                const start = Number(oldState.start || 0);
                const count = Number(oldState.count || 0);
                const warned = !!oldState.warned;

                if (start > 0 && (now - start) <= RATE_LIMIT_WINDOW_MS) {
                    nextState = { start, count: count + 1, warned, touched: now };
                }
            }

            let shouldWarn = false;
            if (nextState.count > RATE_LIMIT_MAX && !nextState.warned) {
                nextState.warned = true;
                shouldWarn = true;
            }

            RATE_STATE_CACHE.set(key, nextState);

            if (nextState.count <= RATE_LIMIT_MAX) {
                return false;
            }

            if (shouldWarn) {
                ctx.waitUntil(tg("sendMessage", {
                    chat_id: uid,
                    text: UI_TEXT.GUEST_RATE_LIMIT
                }));
            }

            return true;
        }

        function formatErrTime() {
            const parts = new Intl.DateTimeFormat("en-CA", {
                timeZone: "Asia/Taipei",
                year: "numeric",
                month: "2-digit",
                day: "2-digit",
                hour: "2-digit",
                minute: "2-digit",
                second: "2-digit",
                hour12: false
            }).formatToParts(new Date());

            const map = {};
            for (const p of parts) {
                if (p.type !== "literal") map[p.type] = p.value;
            }
            return `${map.year}-${map.month}-${map.day} ${map.hour}:${map.minute}:${map.second} Asia/Taipei`;
        }

        async function pushErrorLog(level, text) {
            const firstLine = String(text || "未知错误").split("\n")[0].slice(0, 220);
            const entry = `[${formatErrTime()}] ${level} ${firstLine}`;

            try {
                await KV.put(errorLogEntryKey(level, text), entry, { expirationTtl: TTL.ERRLOG });
            } catch (e) {
                console.error("错误滚动日志写入失败", e);
            }
        }

        async function getRecentErrorLogs() {
            try {
                const keys = [];
                let cursor = undefined;
                let pages = 0;

                while (true) {
                    pages++;
                    if (pages > COUNTALL_MAX_PAGES) break;

                    const page = await KV.list({ prefix: errorLogPrefix(), cursor });
                    for (const item of page.keys || []) {
                        if (item?.name) keys.push(String(item.name));
                    }

                    if (page.list_complete) break;
                    cursor = page.cursor;
                    if (!cursor) break;
                }

                const recentKeys = keys.sort().reverse().slice(0, 5);
                const values = await Promise.all(recentKeys.map((key) => KV.get(key)));
                return values.filter(Boolean).map((value) => String(value));
            } catch (e) {
                console.error("错误滚动日志读取失败", e);
                return [];
            }
        }

        async function countAll(prefix) {
            let total = 0;
            let cursor = undefined;
            let pages = 0;

            while (true) {
                pages++;
                if (pages > COUNTALL_MAX_PAGES) {
                    console.warn(`countAll page cap reached for prefix=${prefix}`);
                    break;
                }

                const page = await KV.list({ prefix, cursor });
                total += (page.keys || []).length;

                if (page.list_complete) break;
                cursor = page.cursor;
                if (!cursor) break;
            }

            return total;
        }

        async function buildAdminStatusText() {
            const banCount = await countAll(KV_PREFIX.BAN);
            let lastErr = "暂无记录";
            try {
                const recentLogs = await getRecentErrorLogs();
                lastErr = recentLogs[0] || "暂无记录";
            } catch (e) {
                // best-effort
            }

            return `🟢 RelayBot 运行正常
🔖 版本：${VERSION}
📦 主线：Topic 包厢模式
🚫 黑名单：${banCount} 人
⚠️ 最终报错：${lastErr}`;
        }

        async function buildAdminErrorsText() {

        function parseAdminAnchorRecord(raw) {
            if (!raw) return null;

            try {
                const data = JSON.parse(raw);
                const messageId = Number(data?.message_id || 0);
                if (!messageId) return null;

                return {
                    message_id: messageId,
                    created_at: Number(data?.created_at || 0) || 0
                };
            } catch (e) {
                const messageId = Number(raw || 0);
                return messageId ? { message_id: messageId, created_at: 0 } : null;
            }
        }

        async function putAdminAnchorRecord(messageId) {
            if (!messageId) return;
            try {
                await KV.put(adminAnchorKey(), JSON.stringify({
                    message_id: Number(messageId),
                    created_at: Date.now()
                }));
            } catch (e) {
                // best-effort
            }
        }

        async function recreateAdminAnchor() {
            try {
                const oldRecord = parseAdminAnchorRecord(await KV.get(adminAnchorKey()));
                if (oldRecord?.message_id) {
                    await retireAdminPanelMessage(oldRecord.message_id);

                // best-effort

        async function ensureAdminAnchor() {
            try {
                const raw = await KV.get(adminAnchorKey());
                if (!parseAdminAnchorRecord(raw)) {
                    return sendAdminAnchor();
                }
            } catch (e) {
                // best-effort
            }
            return { ok: true };

        async function isUsablePanelCallback(cq) {
            const messageId = Number(cq?.message?.message_id || 0);
            if (!messageId) return true;

            let current = null;
            try {
                current = parseAdminAnchorRecord(await KV.get(adminAnchorKey()));
            } catch (e) {
                current = null;
            }

            if (!current?.message_id) {
                await putAdminAnchorRecord(messageId);
                return true;
            }

            if (Number(current.message_id) === messageId) {
                return true;
            }

            await retireAdminPanelMessage(messageId);
            ctx.waitUntil(tg("answerCallbackQuery", {
                callback_query_id: cq.id,
                text: "这是旧控制面板，请使用最新面板。",
                show_alert: true
            }));
            return false;
        }

        const AlertBus = {
            coolKey(dedupeKey) {
                return KV_PREFIX.SYS + "cd_" + dedupeKey;
            },
            inflightKey(dedupeKey) {
                return KV_PREFIX.SYS + "if_" + dedupeKey;
            },
            resolveLevel(text, isCore) {
                if (String(text).startsWith("🚨 RelayBot 主链路崩溃重试")) return "CORE";
                if (String(text).startsWith("🚨 RelayBot 终端熔断告警")) return "TERMINAL";
                if (String(text).startsWith("⚠️ RelayBot 系统降级告警")) return "WARN";
                if (String(text).startsWith("🚨 RelayBot 未知系统崩溃")) return "UNKNOWN";
                return isCore ? "CORE" : "WARN";
            },
            async recordLast(text, isCore) {
                await pushErrorLog(this.resolveLevel(text, isCore), text);
            },
            async emit(text, isCore) {
                await this.recordLast(text, isCore);

                const prefix = (text.startsWith("🚨") || text.startsWith("⚠️"))
                    ? ""
                    : `🚨 RelayBot 告警 (${isCore ? "主链路" : "降级"})\n\n`;

                const res = await tg("sendMessage", {
                    chat_id: MY_ID,
                    text: prefix + text
                });

                if (!res.ok) {
                    throw new Error(`AlertBus emit failed: ${res.description || "unknown"}`);
                }
            },
            dispatch(text, dedupeKey, isCore = false) {
                const job = (async () => {
                    const cdKey = this.coolKey(dedupeKey);
                    const ifKey = this.inflightKey(dedupeKey);
                    let inflightLocked = false;

                    try {
                        const [hasCooldown, hasInflight] = await Promise.all([KV.get(cdKey), KV.get(ifKey)]);
                        if (hasCooldown || hasInflight) return;
                    } catch (e) {
                        console.error("AlertBus precheck failed", e);
                    }

                    try {
                        await KV.put(ifKey, "1", { expirationTtl: TTL.INFLIGHT });
                        inflightLocked = true;
                    } catch (e) {
                        console.error("AlertBus inflight lock failed", e);
                    }

                    try {
                        await this.emit(text, isCore);

                        try {
                            await KV.put(cdKey, "1", { expirationTtl: TTL.COOLDOWN });
                        } catch (e) {
                            console.error("AlertBus cooldown write failed", e);
                        }
                    } catch (e) {
                        console.error("AlertBus emit failed", e);
                    } finally {
                        if (inflightLocked) {
                            try {
                                await KV.delete(ifKey);
                            } catch (e) {
                                console.error("AlertBus inflight unlock failed", e);
                            }
                        }
                    }
                })();

                ctx.waitUntil(job);
            }
        };

                            AlertBus.dispatch(
                                `访客(UID:${uid})拒收冷却锁失败: ${err.message}`,
                                `remind_fail_${uid}`
                            );
                        }
                        return;
                    }

                    const verifiedRaw = await KV.get(KV_PREFIX.VERIFY + uid);
                    const verifiedState = parseVerifyState(verifiedRaw);

                    if (isCommand(cmd, GUEST_COMMANDS.START)) {
                        if (isVerifiedState(verifiedState)) {
                            if (!verifiedState.donePrompted) {
                                try {
                                    await KV.put(KV_PREFIX.VERIFY + uid, serializeVerifyState({
                                        ok: true,
                                        donePrompted: true
                                    }));
                                } catch (err) {
                                    AlertBus.dispatch(
                                        `已验证提示状态(UID:${uid})写入失败: ${err.message}`,
                                        `vdone_${uid}`,
                                        false
                                    );
                                }

                                await tg("sendMessage", {
                                    chat_id: uid,
                                    text: UI_TEXT.VERIFY_DONE
                                });
                            }
                            return;
                        }

                        const waitingRaw = await KV.get(KV_PREFIX.VWAIT + uid);
                        let waiting = parseVerifyWaitState(waitingRaw);

                        if (isVerifyWaitActive(waiting) && waiting?.code) {
                            if (!waiting.startWarned) {
                                waiting.startWarned = true;
                                try {
                                    await KV.put(KV_PREFIX.VWAIT + uid, serializeVerifyWaitState(waiting), {
                                        expirationTtl: verifyWaitTtl()
                                    });
                                } catch (err) {
                                    AlertBus.dispatch(
                                        `验证二次提醒状态(UID:${uid})写入失败: ${err.message}`,
                                        `vstart_${uid}`,
                                        false
                                    );
                                }

                                await tg("sendMessage", {
                                    chat_id: uid,
                                    text: UI_TEXT.VERIFY_START_WAIT
                                });
                            }
                            return;
                        }


                        clearNeedStartWarn(uid);

                        const code = makeVerifyCode();
                        waiting = buildVerifyWaitState(code);

                        try {
                            await KV.put(KV_PREFIX.VWAIT + uid, serializeVerifyWaitState(waiting), {
                                expirationTtl: verifyWaitTtl()
                            });
                        } catch (err) {
                            throw new Error(`[CORE] 验证窗口(UID:${uid})创建失败: ${err.message}`);
                        }

                        const verifyPromptMsg = verifyTextWithCode(UI_TEXT.VERIFY_PROMPT(waiting.code), waiting.code);
                        const verifyPromptRes = await tg("sendMessage", {
                            chat_id: uid,
                            text: verifyPromptMsg.text,
                            entities: verifyPromptMsg.entities
                        });
                        if (!verifyPromptRes.ok) {
                            throw new Error(`[CORE] 验证码(UID:${uid})发送失败: ${verifyPromptRes.description || "unknown"}; verify code send failed`);
                        }
                        return;
                    }

                    if (cmd && !isCommand(cmd, GUEST_COMMANDS.START)) {
                        return;
                    }

                    if (!isVerifiedState(verifiedState)) {
                        const waitingRaw = await KV.get(KV_PREFIX.VWAIT + uid);
                        const waiting = parseVerifyWaitState(waitingRaw);

                        if (waitingRaw && waiting?.code) {
                            if (!isVerifyWaitActive(waiting)) {
                                const expiredWarnCount = Math.max(0, Number(waiting.expiredWarnCount || 0) || 0);
                                if (expiredWarnCount < 1) {
                                    waiting.expiredWarnCount = expiredWarnCount + 1;
                                    try {
                                        await KV.put(KV_PREFIX.VWAIT + uid, serializeVerifyWaitState(waiting), {
                                            expirationTtl: verifyWaitTtl()
                                        });
                                    } catch (err) {
                                        AlertBus.dispatch(
                                            `验证码过期提醒计数(UID:${uid})写入失败: ${err.message}`,
                                            `vexpired_${uid}`,
                                            false
                                        );
                                    }

                                    await tg("sendMessage", {
                                        chat_id: uid,
                                        text: UI_TEXT.VERIFY_EXPIRED
                                    });
                                }
                                return;
                            }

                            if (normalizeVerifyCode(rawText) === waiting.code) {
                                try {
                                    await KV.put(KV_PREFIX.VERIFY + uid, serializeVerifyState({
                                        ok: true,
                                        donePrompted: false
                                    }));
                                } catch (err) {
                                    throw new Error(`[CORE] 验证状态(UID:${uid})落盘失败: ${err.message}`);
                                }

                                clearNeedStartWarn(uid);

                                await tg("sendMessage", {
                                    chat_id: uid,
                                    text: UI_TEXT.VERIFY_OK
                                });
                                return;
                            }

                            const wrongWarnCount = Math.max(0, Number(waiting.wrongWarnCount || 0) || 0);
                            if (wrongWarnCount < 2) {
                                waiting.wrongWarnCount = wrongWarnCount + 1;
                                try {
                                    await KV.put(KV_PREFIX.VWAIT + uid, serializeVerifyWaitState(waiting), {
                                        expirationTtl: verifyWaitTtl()
                                    });
                                } catch (err) {
                                    AlertBus.dispatch(
                                        `验证码错误提醒计数(UID:${uid})写入失败: ${err.message}`,
                                        `vwrong_${uid}`,
                                        false
                                    );
                                }

                                const verifyWrongMsg = verifyTextWithCode(UI_TEXT.VERIFY_WRONG(waiting.code), waiting.code);
                                await tg("sendMessage", {
                                    chat_id: uid,
                                    text: verifyWrongMsg.text,
                                    entities: verifyWrongMsg.entities
                                });
                            }
                            return;
                        }

                        if (shouldSendNeedStart(uid)) {
                            await tg("sendMessage", {
                                chat_id: uid,
                                text: UI_TEXT.VERIFY_NEED_START
                            });
                        }
                        return;
                    }

                    if (await isGuestRateLimited(uid)) {
                        return;
                    }

                    let threadId = normalizeThreadId(await KV.get(KV_PREFIX.TOPIC + uid));

                    const buildTopic = async () => {
                        const topicKey = KV_PREFIX.TOPIC + uid;
                        const lockKey = topicLockKey(uid);

                        const existedBefore = normalizeThreadId(await KV.get(topicKey));
                        if (existedBefore) {
                            return existedBefore;
                        }

                        const lockSeen = await KV.get(lockKey);
                        if (lockSeen) {
                            await sleep(TOPIC_LOCK_WAIT_MS);
                            const waitedTid = normalizeThreadId(await KV.get(topicKey));
                            if (waitedTid) {
                                return waitedTid;
                            }
                        }

                        try {
                            await KV.put(lockKey, "1", { expirationTtl: TOPIC_LOCK_TTL });
                        } catch (err) {
                            AlertBus.dispatch(
                                `建房软锁(UID:${uid})写入失败: ${err.message}`,
                                `topic_lock_${uid}`
                            );
                        }

                        const existedAfterLock = normalizeThreadId(await KV.get(topicKey));
                        if (existedAfterLock) {
                            return existedAfterLock;
                        }

                        const rawName = String(msg.from?.first_name || "User").replace(/\s+/g, " ").trim() || "User";
                        const topicName = `${rawName} | ${uid}`.slice(0, 128);

                        const tr = await tg("createForumTopic", {
                            chat_id: MY_ID,
                            name: topicName
                        });

                        if (!tr.ok) {
                            const topicErr = parseErr(tr);
                            if (topicErr.action === "CORE") {
                                throw new Error(`[CORE] 建房(UID:${uid})失败: ${topicErr.desc}`);
                            }
                            if (topicErr.action === "TERMINAL") {
                                throw new Error(`[TERMINAL] 建房(UID:${uid})失败: ${topicErr.desc}`);
                            }
                            throw new Error(`[CORE] 建房(UID:${uid})失败: ${topicErr.desc}`);
                        }

                        const newTid = normalizeThreadId(tr.result?.message_thread_id);
                        if (!newTid) {
                            throw new Error(`[CORE] 建房(UID:${uid})未返回有效 Topic ID`);
                        }

                        const cleanupLeakedTopic = async () => {
                            try {
                                await KV.delete(KV_PREFIX.THREAD + newTid);
                            } catch (e) {
                                // best-effort
                            }
                            await deleteTopicNow(newTid);
                        };

                        const checkOld = normalizeThreadId(await KV.get(KV_PREFIX.TOPIC + uid));
                        if (checkOld) {
                            ctx.waitUntil(cleanupLeakedTopic());
                            return checkOld;
                        }

                        try {
                            await Promise.all([
                                KV.put(KV_PREFIX.TOPIC + uid, newTid),
                                KV.put(KV_PREFIX.THREAD + newTid, uid)
                            ]);
                        } catch (err) {
                            throw new Error(`[CORE] 建房映射(UID:${uid})落盘失败: ${err.message}`);
                        }

                        const finalTid = normalizeThreadId(await KV.get(KV_PREFIX.TOPIC + uid));
                        if (finalTid && finalTid !== newTid) {
                            ctx.waitUntil(cleanupLeakedTopic());
                            return finalTid;
                        }

                        return newTid;
                    };

                    if (threadId) {
                        const reverseUid = await KV.get(KV_PREFIX.THREAD + threadId);

                        if (String(reverseUid || "") === uid) {
                            // 正常双向映射，直接使用
                        } else if (!reverseUid) {
                            const cutover = await KV.get(orphanFixMarkKey(threadId));

                            if (cutover) {
                                try {
                                    await Promise.all([
                                        KV.delete(KV_PREFIX.TOPIC + uid),
                                        KV.delete(orphanFixMarkKey(threadId))
                                    ]);
                                } catch (err) {
                                    throw new Error(`[CORE] 切换标记(UID:${uid})清理失败: ${err.message}`);
                                }
                                threadId = null;
                            } else {
                                try {
                                    await KV.put(KV_PREFIX.THREAD + threadId, uid);
                                } catch (err) {
                                    throw new Error(`[CORE] 反向映射(UID:${uid},Topic:${threadId})补建失败: ${err.message}`);
                                }
                            }
                        } else {
                            try {
                                await KV.delete(KV_PREFIX.TOPIC + uid);
                            } catch (err) {
                                throw new Error(`[CORE] 冲突映射(UID:${uid})清理失败: ${err.message}`);
                            }
                            threadId = null;
                        }
                    }

                    let topicJustCreated = false;
                    if (!threadId) {
                        threadId = normalizeThreadId(await buildTopic());
                        topicJustCreated = true;
                    }

                    if (!isValidThreadId(threadId)) {
                        throw new Error(`[CORE] 投递访客(UID:${uid})前缺少有效 Topic ID，已阻止消息落入总控制室`);
                    }

                    if (topicJustCreated) {
                        const ready = await ensureTopicReady(threadId);
                        if (!ready) {
                            throw new Error(`[CORE] 新建包厢(UID:${uid},Topic:${threadId})就绪探测失败`);
                        }
                    }

                    const fwdParams = withRequiredThread({
                        chat_id: MY_ID,
                        from_chat_id: uid,
                        message_id: msg.message_id
                    }, threadId, `投递访客(UID:${uid})`);

                    const fwd = await tg("forwardMessage", fwdParams);
                    await assertForwardLandedInThread(fwd, threadId, uid, "投递访客");

                    if (!fwd.ok) {
                        const err = parseErr(fwd);

                        if (err.action === "HEAL") {
                            try {
                                await Promise.all([
                                    KV.delete(KV_PREFIX.TOPIC + uid),
                                    threadId ? KV.delete(KV_PREFIX.THREAD + threadId) : Promise.resolve(),
                                    threadId ? KV.delete(orphanFixMarkKey(threadId)) : Promise.resolve()
                                ]);
                            } catch (err2) {
                                throw new Error(`[CORE] 死链映射(UID:${uid})清理失败: ${err2.message}`);
                            }

                            const retryTid = normalizeThreadId(await buildTopic());
                            if (!isValidThreadId(retryTid)) {
                                throw new Error(`[CORE] 自愈建房(UID:${uid})未得到有效 Topic ID，已阻止消息落入总控制室`);
                            }
                            const retryReady = await ensureTopicReady(retryTid);
                            if (!retryReady) {
                                throw new Error(`[CORE] 自愈建房(UID:${uid},Topic:${retryTid})就绪探测失败`);
                            }

                            const retryFwd = await tg("forwardMessage", withRequiredThread({
                                chat_id: MY_ID,
                                from_chat_id: uid,
                                message_id: msg.message_id
                            }, retryTid, `自愈重发(UID:${uid})`));
                            await assertForwardLandedInThread(retryFwd, retryTid, uid, "自愈重发");

                            if (!retryFwd.ok) {
                                const retryErr = parseErr(retryFwd);

                                if (retryErr.action === "CORE") {
                                    throw new Error(`[CORE] 自愈重发(UID:${uid})失败: ${retryErr.desc}`);
                                }

                                if (retryErr.action === "TERMINAL") {
                                    throw new Error(`[TERMINAL] 自愈重发(UID:${uid})被拒: ${retryErr.desc}`);
                                }

                                AlertBus.dispatch(
                                    `自愈重投(UID:${uid})降级失败: ${retryErr.desc}`,
                                    `heal_${uid}`
                                );
                                throw new Error(`[CORE] 自愈重发(UID:${uid})失败: ${retryErr.desc}; heal retry forward not delivered`);
                            }

                            ctx.waitUntil((async () => {
                                try {
                                    await tg("unpinAllForumTopicMessages", {
                                        chat_id: MY_ID,
                                        message_thread_id: Number(retryTid)
                                    });
                                } catch (e) {
                                    // best-effort
                                }
                            })());
                            return;
                        }

                        if (err.action === "CORE") {
                            throw new Error(`[CORE] 投递访客(UID:${uid})消息失败: ${err.desc}`);
                        }

                        if (err.action === "TERMINAL") {
                            throw new Error(`[TERMINAL] 投递访客(UID:${uid})被拒: ${err.desc}`);
                        }

                        AlertBus.dispatch(
                            `投递(UID:${uid})降级失败: ${err.desc}`,
                            `fwd_in_${uid}`
                        );
                    }

                    if (topicJustCreated && fwd.ok) {
                        ctx.waitUntil((async () => {
                            try {
                                await tg("unpinAllForumTopicMessages", {
                                    chat_id: MY_ID,
                                    message_thread_id: Number(threadId)
                                });
                            } catch (e) {
                                // best-effort
                            }
                        })());
                    }
                }
            };

            await processUpdate();

            try {
                if (b.update_id !== undefined) {
                    await KV.put(KV_PREFIX.UPD + b.update_id, "1", { expirationTtl: TTL.UPD });
                }
            } catch (err) {
                AlertBus.dispatch(
                    `${UI_TEXT.ERR_UPD}: ${err.message}`,
                    `upd_${b.update_id}`
                );
            } finally {
                if (b.update_id !== undefined) {
                    setHotUpdateState(b.update_id, "done");
                }
            }

        } catch (e) {
            if (b?.update_id !== undefined) {
                clearHotUpdateState(b.update_id);
            }

            console.error("RelayBot V1.0 捕获:", e);
            const errStr = String(e?.stack || e?.message || e);
            const dedupeKey = "out_" + hashStr(errStr);

            if (BOT_TOKEN && MY_ID) {
                if (errStr.includes("[CORE]")) {
                    AlertBus.dispatch(
                        `🚨 RelayBot 主链路崩溃重试\n\n异常详情：${errStr.substring(0, 500)}`,
                        dedupeKey,
                        true
                    );
                    return new Response("Core Execution Failed", { status: 500 });
                }

                if (errStr.includes("[TERMINAL]")) {
                    AlertBus.dispatch(
                        `🚨 RelayBot 终端熔断告警 (不重试)\n\n异常详情：${errStr.substring(0, 500)}`,
                        dedupeKey,
                        false
                    );
                    return new Response("OK");
                }

                if (errStr.includes("[WARN]")) {
                    AlertBus.dispatch(
                        `⚠️ RelayBot 系统降级告警\n\n异常详情：${errStr.substring(0, 500)}`,
                        dedupeKey,
                        false
                    );
                    return new Response("OK");
                }

                AlertBus.dispatch(
                    `🚨 RelayBot 未知系统崩溃 (触发保底重试)\n\n捕获到未分类错误：\n${errStr.substring(0, 500)}`,
                    dedupeKey,
                    true
                );
                return new Response("Unknown System Error", { status: 500 });
            }

            return new Response("Error", { status: 500 });
        }

        return new Response("OK");
    }
};
